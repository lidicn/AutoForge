### `src/autoforge/af_expect.py`（273 行 / 原 10,011 → 内联 10,011 字符，快照 `12,180B sha256:b93058d7afc4 @2026-09-17T19:56:14`，**逻辑原样照抄，不许改**）

```python
"""v1.2.0 后置条件断言求值（`expect`）—— 让 `forge sim` 能回答「跑**对**了吗」。

**为什么需要它**：在 v1.2.0 之前，`forge sim` 只返回 `final_states`——**跑完了，但没人说「应该是什么」**。
于是 `forge build` 拦住了「不安全」，`forge sim` **没拦住「逻辑对但语义错」**。

本模块把 IR 顶层的 `expect` 逐条求值，产出可机读也可人读的三态结果：

| 状态 | 含义 |
|---|---|
| `pass` | 断言成立 |
| `fail` | 断言不成立（**自动化没跑对**） |
| `unverified` | **无法验证**（实体不在状态源 / 变量没被任何实例写过 / 无法比较） |

> **`unverified` 必须与 `pass` 区分开**——把「没验到」当「验过了」，断言就成了自欺。
> 这与 `af_vhass` 对「未建模服务」的处理是同一条诚实性主线（不伪造、只如实降级）。

`expect` 三种形态：
- **实体形态** `{"entity_id": "light.study_main", "state": "on"}`（`state` 可为候选数组，命中任一即过）；
- **属性形态**（v1.7.1）`{"entity_id": "climate.ac", "attribute": "temperature", "value": 18}`——
  断言的是**属性**而非状态（`op` 缺省 `eq`，支持 `lt/gt` 等）。
  为什么必须有：HA 里「设定温度 / 亮度 / 当前活动」都在 attributes 里，
  而 `state` 是 `cool`/`on` 这类粗粒度值。NL 实测 #3 的军令状是「回调到 **18 度**」，
  用 `state: ["off","cool",...]` 断言时**播种值就已满足**——断言恒真，
  等于没验证（vacuous assertion）。属性形态才能真回答「跑对了吗」。
- **变量形态** `{"var": "turn_on_result.success", "op": "eq", "value": true}`（`op` 缺省 `eq`）。
"""

from __future__ import annotations

from typing import Any, Iterable, Mapping, Sequence

__all__ = ["ExpectReport", "evaluate_expects", "evaluate_graph_expects", "resolve_var"]

#: 哨兵：变量不存在（与「变量存在但值为 None」区分）
MISSING: Any = object()

#: 变量形态支持的前缀（与 `af_ir.expr` 的命名空间一致）
_VAR_PREFIXES = ("vars.", "context.", "entity.")


def resolve_var(name: str, namespace: Mapping[str, Any]) -> Any:
    """按点号路径解析变量：`turn_on_result.success` / `vars.flag`。不存在返回 `MISSING`。"""
    path = str(name or "")
    for prefix in _VAR_PREFIXES:
        if path.startswith(prefix):
            path = path[len(prefix) :]
            break
    current: Any = namespace
    for part in path.split("."):
        if isinstance(current, Mapping) and part in current:
            current = current[part]
        else:
            return MISSING
    return current


def _apply_op(op: str, actual: Any, expected: Any) -> bool | None:
    """比较运算。返回 `None` 表示「无法比较」（→ unverified，而非判失败）。"""
    if op == "eq":
        if isinstance(actual, str) or isinstance(expected, str):
            return str(actual) == str(expected)
        return actual == expected
    if op == "ne":
        inner = _apply_op("eq", actual, expected)
        return None if inner is None else (not inner)
    try:
        left, right = float(actual), float(expected)
    except (TypeError, ValueError):
        return None
    if op == "lt":
        return left < right
    if op == "lte":
        return left <= right
    if op == "gt":
        return left > right
    if op == "gte":
        return left >= right
    return None


def _state_matches(actual: Any, expected: Any) -> bool:
    """实体形态：`state` 可为字符串或候选数组（命中任一即通过）。"""
    if isinstance(expected, (list, tuple)):
        return any(str(actual) == str(x) for x in expected)
    return str(actual) == str(expected)


def _read_attribute(states: Any, entity_id: str, attribute: str) -> Any:
    """从状态源读实体属性；读不到返回 `MISSING`（→ unverified，绝不编值）。

    优先走 `snapshot([entity_id])`（与生产 `HAStateProvider` / 仿真 `FakeHA` 同一入口），
    状态源不支持快照时回退直读 `.attributes`。
    """
    snapshot_fn = getattr(states, "snapshot", None)
    if callable(snapshot_fn):
        try:
            snapshot = snapshot_fn([entity_id])
            attrs = getattr(snapshot, "attributes", None) or {}
            got = attrs.get(entity_id) or {}
            if isinstance(got, Mapping) and attribute in got:
                return got[attribute]
            return MISSING
        except Exception:  # 状态源不可达（生产网络/鉴权）→ 落回直读，最终 unverified
            pass
    direct = getattr(states, "attributes", None)
    if isinstance(direct, Mapping):
        got = direct.get(entity_id) or {}
        if isinstance(got, Mapping) and attribute in got:
            return got[attribute]
    return MISSING


class ExpectReport(dict):
    """断言报告：可直接 JSON 化的 dict（`ok` / `checked` / `passed` / `failed` / `unverified` / `items`）。"""


def evaluate_expects(
    auto: Any,
    states: Any,
    var_sources: Sequence[Mapping[str, Any]] | None = None,
) -> ExpectReport:
    """对单条自动化的 `expect` 逐条求值。

    - `states`：状态源（实现 `.get(entity_id) -> str | None`，与 `af_service.simulate` 同源）。
    - `var_sources`：若干「变量命名空间」（通常每个实例一份 `ctx.vars`）；
      变量形态断言**在任一命名空间中成立即通过**。为空表示没有活跃实例 → `unverified`。
    """
    namespaces = list(var_sources or ())
    items: list[dict[str, Any]] = []
    passed = failed = unverified = 0

    for index, raw in enumerate(auto.expects()):
        item: dict[str, Any] = {"index": index, "note": raw.get("note", "")}
        if raw.get("entity_id"):
            entity_id = str(raw["entity_id"])
            attribute = raw.get("attribute")
            if attribute:
                # ── 属性形态（v1.7.1）────────────────────────────────────
                attr = str(attribute)
                op = str(raw.get("op") or "eq")
                expected = raw.get("value")
                actual = _read_attribute(states, entity_id, attr)
                item.update(
                    {
                        "kind": "entity_attribute",
                        "target": f"{entity_id}.{attr}",
                        "op": op,
                        "expected": expected,
                        "actual": None if actual is MISSING else actual,
                    }
                )
                if actual is MISSING:
                    item.update(
                        status="unverified",
                        reason=(
                            f"实体 {entity_id} 的 {attr!r} 属性不在状态源中"
                            "（未被播种或设备未上报），本次仿真无法验证该断言"
                        ),
                    )
                    unverified += 1
                else:
                    outcome = _apply_op(op, actual, expected)
                    if outcome is True:
                        item["status"] = "pass"
                        passed += 1
                    elif outcome is None:
                        item.update(status="unverified", reason="无法比较（类型不可比）")
                        unverified += 1
                    else:
                        item["status"] = "fail"
                        item["reason"] = f"期望 {op} {expected!r}，实际 {actual!r}"
                        failed += 1
            else:
                expected = raw.get("state")
                actual = states.get(entity_id) if hasattr(states, "get") else None
                item.update(
                    {"kind": "entity", "target": entity_id, "expected": expected, "actual": actual}
                )
                if actual is None:
                    item.update(
                        status="unverified",
                        reason="实体不在状态源中（未被播种或不可达），本次仿真无法验证该断言",
                    )
                    unverified += 1
                elif _state_matches(actual, expected):
                    item["status"] = "pass"
                    passed += 1
                else:
                    item["status"] = "fail"
                    item["reason"] = f"期望 {expected!r}，实际 {actual!r}"
                    failed += 1
        else:
            var_name = str(raw.get("var", ""))
            op = str(raw.get("op") or "eq")
            expected = raw.get("value")
            values = [resolve_var(var_name, ns) for ns in namespaces]
            present = [v for v in values if v is not MISSING]
            item.update({"kind": "var", "target": var_name, "op": op, "expected": expected})

            if not namespaces:
                item.update(status="unverified", reason="没有活跃实例（仿真未产生任何实例），无法验证变量断言")
                unverified += 1
                continue
            if not present:
                item.update(status="unverified", reason=f"没有任何实例写过变量 {var_name!r}")
                unverified += 1
                continue

            outcomes = [_apply_op(op, v, expected) for v in present]
            item["actual"] = values[0] if len(values) == 1 else present
            if any(o is True for o in outcomes):
                item["status"] = "pass"
                passed += 1
            elif all(o is None for o in outcomes):
                item.update(status="unverified", reason="无法比较（类型不可比）")
                unverified += 1
            else:
                item["status"] = "fail"
                item["reason"] = f"期望 {op} {expected!r}，实际 {present!r}"
                failed += 1

        items.append(item)

    return ExpectReport(
        ok=failed == 0,
        #: `fully_verified` 比 `ok` 更严：声明过断言、且**全部验过**（无 fail 且无 unverified）。
        #: `ok=True` 只代表「没抓到反例」——把「没验到」当「验过了」就是自欺（对齐 autoflow 的 fully_verified 语义）。
        fully_verified=bool(items) and failed == 0 and unverified == 0,
        declared=len(items),
        checked=passed + failed,
        passed=passed,
        failed=failed,
        unverified=unverified,
        items=items,
    )


def evaluate_graph_expects(
    graph: Any,
    states: Any,
    var_sources: Sequence[Mapping[str, Any]] | None = None,
    unmodeled_actions: Iterable[str] | None = None,
) -> dict[str, Any]:
    """对整个 Graph 求值：按自动化聚合。**未声明 `expect` 的自动化不进入结果**。

    `unmodeled_actions`：仿真底座未建模的动作（如 vhass/FakeHA 不认识的服务）——
    有则 `fully_verified=False`（那些动作的后果根本没被验证过）。
    """
    per_auto: dict[str, Any] = {}
    total = {"declared": 0, "passed": 0, "failed": 0, "unverified": 0}
    for auto in graph:
        if not auto.expects():
            continue
        report = evaluate_expects(auto, states, var_sources)
        per_auto[auto.id] = report
        for key in total:
            total[key] += report.get(key, 0)
    unmodeled = sorted(set(unmodeled_actions or ()))
    return {
        "ok": total["failed"] == 0,
        "fully_verified": (
            total["declared"] > 0
            and total["failed"] == 0
            and total["unverified"] == 0
            and not unmodeled
        ),
        "declared": total["declared"],
        "passed": total["passed"],
        "failed": total["failed"],
        "unverified": total["unverified"],
        "unmodeled_actions": unmodeled,
        "automations": per_auto,
    }

```

class SimulateReport(dict):
    """`simulate` 诚实分层报告：可直接 JSON 化的 dict。

    顶层 = `evaluate_graph_expects` 的原字段（保持不变，向后兼容）+ `report_version` +
    `final_states` + 三个分区字段（`SECTIONS`）+ `section_counts`。
    """

    def sections(self) -> dict[str, list[dict[str, Any]]]:
        """三个分区（键恒齐全、值恒为 list，可直接 `for entry in ...` 枚举）。"""
        return {name: list(self.get(name) or ()) for name in SECTIONS}

    def flagged(self, flag: str = FLAG_YELLOW) -> list[dict[str, Any]]:
        """所有带指定 `flag` 的条目（默认：标黄的不可仿真条目）。"""
        out: list[dict[str, Any]] = []
        for name in SECTIONS:
            for entry in self.get(name) or ():
                if entry.get("flag") == flag:
                    out.append(entry)
        return out
```

    """只读提取 `(entity_id -> 最终状态)` 观测面，供 `inferred` 分区使用。**绝不写回、绝不编值**。

    优先 `final_states`（`forge sim` 的既有产物），否则从状态源探测：`items()` → `.attributes` 键；
    只探测到键、读不到值时保持 `None`（如实，不猜值）。

    observed: dict[str, Any] = {}
    for source in (final_states,):
        if isinstance(source, Mapping):
            for key in source:
                observed[str(key)] = _as_state(source[key])
    items_fn = getattr(states, "items", None)
    if callable(items_fn):

                observed.setdefault(str(key), _as_state(value))
        except Exception:
            pass
    attributes = getattr(states, "attributes", None)
    if isinstance(attributes, Mapping):

    getter = getattr(states, "get", None)
    if callable(getter):

                try:
                    got = getter(entity_id)
                except Exception:
                    got = None
                if got is not None:

class _SnapshotStates(_States):
    """支持 `.snapshot()` 的状态源；`raises=True` 模拟生产网络/鉴权不可达（应降级为 unverified）。"""

        if self._raises:
            raise RuntimeError("state source unreachable")
        return SimpleNamespace(
            attributes={eid: self._snapshot_attributes.get(eid, {}) for eid in entity_ids}

# 一、设计定案（先定方向，再写代码）

**分层轴 = 证据强度，不是结论好坏。** 三栏回答的是「这条结论我们凭什么信」：

| 分区 | 进入条件 | 语义 | flag |
|---|---|---|---|
| `verified` | 断言**真跑过**（`pass` / `fail`） | 有证据。`fail` 是**验证过的反例**，照样算验过 | `None` |
| `inferred` | 有仿真产出、**无断言覆盖**（未声明 expect 的自动化 / 未被断言的最终状态） | 只能推断「大概对了」 | `None` |
| `non_simulable` | 断言 `unverified` / 底座未建模动作 / 契约外状态 | **不可仿真**，显式标黄 | `"yellow"` |

- 三条红线（各配单测）：`unverified` 不许进 `verified`、不许降级成 `inferred`；不可仿真必须 `flag="yellow"`；内核零脚本执行，报告落盘前统一清洗脚本逃生舱标识（fail-closed）。
- **旧字段逐字不变**：`simulate` 返回 = `evaluate_graph_expects` 原字段（`ok / fully_verified / declared / passed / failed / unverified / unmodeled_actions / automations`）+ 新增 `verified / inferred / non_simulable / section_counts / final_states / report_version`（超集 + 版本号双保险）。
- **执行边界**：`simulate` 是纯报告构建点，不执行任何动作、不写盘、不碰真机；`af_executor.py` 一个字不动。

def _iter_automations(graph: Any) -> list[Any]:
    """把 Graph 摊平成自动化列表（契约：可迭代、元素有 `.id` / `.expects()`）。"""
    if graph is None:

def _auto_expects(auto: Any) -> list[Any]:
    """读原始 `expect` 声明（只有 `.expects()` 面在 §1 契约内，故只依赖它）。"""
    fn = getattr(auto, "expects", None)
    return list(fn() or ()) if callable(fn) else []


def _covered_entities(graph: Any, automations: Mapping[str, Any]) -> set[str]:
    """被 expect 覆盖到的实体：其结论已在 `verified` / `non_simulable` 有归属，不再进 `inferred`（避免同一条结论双计）。"""
    covered: set[str] = set()
    for auto in _iter_automations(graph):
        for raw in _auto_expects(auto):
            if isinstance(raw, Mapping) and raw.get("entity_id"):
                covered.add(str(raw["entity_id"]))
    if covered:
        return covered
    # 拿不到 graph 时的回退：从已求值条目反推（属性形态 target 是 f"{entity_id}.{attr}"）
    for report in (automations or {}).values():
        for item in (report.get("items") or ()):
            target = str(item.get("target") or "")
            if item.get("kind") == "entity":
                covered.add(target)
            elif item.get("kind") == "entity_attribute" and "." in target:
                covered.add(target.rsplit(".", 1)[0])
    return covered


def _as_state(value: Any) -> Any:
    """状态源里一个实体的值：允许 `{state, attributes}` 结构，取 `state`；否则原样返回。"""
    if isinstance(value, Mapping) and "state" in value:
        return value["state"]
    return value


def _final_state_map(states: Any, final_states: Mapping[str, Any] | None) -> dict[str, Any]:
    """只读提取 `(entity_id -> 最终状态)` 观测面，供 `inferred` 使用。**绝不写回、绝不编值**。

    优先 `final_states`（`forge sim` 既有产物）；否则从状态源探测：`items()` → `.attributes` 键，
    值读不到就保持 `None`（如实，不猜值）。探测异常一律降级为空（只影响 inferred 填充，不影响任何判定）。
    """
    observed: dict[str, Any] = {}
    if isinstance(final_states, Mapping):
        for key in final_states:
            observed[str(key)] = _as_state(final_states[key])
    items_fn = getattr(states, "items", None)
    if callable(items_fn):
        try:
            for key, value in items_fn():
                observed.setdefault(str(key), _as_state(value))
        except Exception:
            pass
    attributes = getattr(states, "attributes", None)
    if isinstance(attributes, Mapping):
        for key in attributes:
            observed.setdefault(str(key), None)
    getter = getattr(states, "get", None)
    if callable(getter):
        for entity_id, value in list(observed.items()):
            if value is None:
                try:
                    got = getter(entity_id)
                except Exception:
                    got = None
                if got is not None:
                    observed[entity_id] = got
    return dict(sorted(observed.items()))


class SimulateReport(dict):
    """`simulate` 诚实分层报告：可直接 JSON 化的 dict。

    顶层 = `evaluate_graph_expects` 的原字段（逐字不变，向后兼容）+ `final_states` +
    `report_version` + 三个分区（`SECTIONS`）+ `section_counts`。
    """

    def sections(self) -> dict[str, list[dict[str, Any]]]:
        """三个分区（键恒齐全、值恒为 list，可直接 `for entry in ...` 枚举）。"""
        return {name: list(self.get(name) or ()) for name in SECTIONS}

    def flagged(self, flag: str = FLAG_YELLOW) -> list[dict[str, Any]]:
        """所有带指定 `flag` 的条目（默认：标黄的不可仿真条目）。"""
        out: list[dict[str, Any]] = []
        for name in SECTIONS:
            for entry in self.get(name) or ():
                if entry.get("flag") == flag:
                    out.append(entry)
        return out


def build_layered_report(
    graph_result: Mapping[str, Any],
    *,
    graph: Any = None,
    states: Any = None,
    final_states: Mapping[str, Any] | None = None,
) -> SimulateReport:
    """把 `evaluate_graph_expects` 的结果按证据强度分成三区（纯函数，不执行任何动作）。

    分派规则（不猜、不美化）：
    - 断言 `pass` / `fail` → `verified`（证据在手，结论好坏由旧字段 `ok` / `failed` 说话）；
    - 断言 `unverified` → `non_simulable` + `flag="yellow"`；
    - 契约外未知状态 → 同样 `non_simulable` 标黄（fail-closed，不猜语义）；
    - 未建模动作 → `non_simulable` + `flag="yellow"`；
    - 未声明 expect 的自动化、未被断言覆盖的最终状态 → `inferred`（只是推断，不许当验证看）。
    """
    verified: list[dict[str, Any]] = []
    inferred: list[dict[str, Any]] = []
    non_simulable: list[dict[str, Any]] = []

    automations: dict[str, Any] = dict(graph_result.get("automations") or {})
    covered = _covered_entities(graph, automations)

    for auto_id, auto_report in automations.items():
        for item in (auto_report.get("items") or ()):
            status = str(item.get("status") or "")
            if status in ("pass", "fail"):
                verified.append(_assertion_entry(str(auto_id), item, status=status, flag=None))
            elif status == "unverified":
                non_simulable.append(
                    _assertion_entry(str(auto_id), item, status="unverified", flag=FLAG_YELLOW)
                )
            else:
                entry = _assertion_entry(str(auto_id), item, status=status or "unknown", flag=FLAG_YELLOW)
                if not entry["reason"]:
                    entry["reason"] = "断言状态不在 af_expect 三态契约内，按不可仿真处理（fail-closed，不猜）"
                non_simulable.append(entry)

    for action in dict.fromkeys(str(a) for a in (graph_result.get("unmodeled_actions") or ())):
        non_simulable.append(
            _entry(
                "unmodeled_action",
                target=action,
                action=action,
                status="unmodeled",
                reason="仿真底座未建模该动作，其后果不可仿真，如实标黄列出",
                flag=FLAG_YELLOW,
            )
        )

    for auto in _iter_automations(graph):
        auto_id = str(getattr(auto, "id", "") or "")
        if not auto_id or auto_id in automations:
            continue
        inferred.append(
            _entry(
                "automation_unasserted",
                auto_id=auto_id,
                target=auto_id,
                status="inferred",
                reason="该自动化未声明 expect，仿真结果没有任何断言覆盖，只能算推断",
                flag=None,
            )
        )

    observed = _final_state_map(states, final_states)
    for entity_id, value in observed.items():
        if entity_id in covered:
            continue
        inferred.append(
            _entry(
                "final_state",
                target=entity_id,
                actual=value,
                status="inferred",
                reason="仿真产出该实体最终状态，但没有 expect 覆盖它，属推断而非验证",
                flag=None,
            )
        )

    payload: dict[str, Any] = dict(graph_result)  # 旧字段原样保留
    payload["final_states"] = dict(final_states) if isinstance(final_states, Mapping) else observed
    payload["report_version"] = REPORT_VERSION
    payload["verified"] = verified
    payload["inferred"] = inferred
    payload["non_simulable"] = non_simulable
    payload["section_counts"] = {name: len(payload[name]) for name in SECTIONS}
    return SimulateReport(_scrub(payload))


def simulate(
    graph: Any,
    states: Any = None,
    var_sources: Sequence[Mapping[str, Any]] | None = None,
    unmodeled_actions: Iterable[str] | None = None,
    final_states: Mapping[str, Any] | None = None,
) -> SimulateReport:
    """构建 `forge sim` 诚实分层报告（**报告构建点**，不执行任何动作、不碰真机）。

    - `graph`：IR 顶层 Graph（可迭代；元素有 `.id` / `.expects()`，与 `evaluate_graph_expects` 同契约）。
    - `states`：状态源（`.get(entity_id)`；可选 `.attributes` / `.snapshot`），与仿真同源。
    - `var_sources`：若干变量命名空间（通常每个实例一份 `ctx.vars`）。
    - `unmodeled_actions`：仿真底座未建模的动作名 → 落 `non_simulable` 标黄。
    - `final_states`：`forge sim` 既有产物（legacy 字段，原样透传）；缺省时从 `states` 只读推导。

    返回 `SimulateReport`：`evaluate_graph_expects` 的原字段 + 三个分区 + `report_version`。
    """
    graph_result = evaluate_graph_expects(graph, states, var_sources, unmodeled_actions)
    return build_layered_report(graph_result, graph=graph, states=states, final_states=final_states)
```

## 2.2 `src/autoforge/af_expect.py`（§1 逐字照抄，一行不改）

```python
"""v1.2.0 后置条件断言求值（`expect`）—— 让 `forge sim` 能回答「跑**对**了吗」。

**为什么需要它**：在 v1.2.0 之前，`forge sim` 只返回 `final_states`——**跑完了，但没人说「应该是什么」**。
于是 `forge build` 拦住了「不安全」，`forge sim` **没拦住「逻辑对但语义错」**。

本模块把 IR 顶层的 `expect` 逐条求值，产出可机读也可人读的三态结果：

| 状态 | 含义 |
|---|---|
| `pass` | 断言成立 |
| `fail` | 断言不成立（**自动化没跑对**） |
| `unverified` | **无法验证**（实体不在状态源 / 变量没被任何实例写过 / 无法比较） |

> **`unverified` 必须与 `pass` 区分开**——把「没验到」当「验过了」，断言就成了自欺。
> 这与 `af_vhass` 对「未建模服务」的处理是同一条诚实性主线（不伪造、只如实降级）。

`expect` 三种形态：
- **实体形态** `{"entity_id": "light.study_main", "state": "on"}`（`state` 可为候选数组，命中任一即过）；
- **属性形态**（v1.7.1）`{"entity_id": "climate.ac", "attribute": "temperature", "value": 18}`——
  断言的是**属性**而非状态（`op` 缺省 `eq`，支持 `lt/gt` 等）。
  为什么必须有：HA 里「设定温度 / 亮度 / 当前活动」都在 attributes 里，
  而 `state` 是 `cool`/`on` 这类粗粒度值。NL 实测 #3 的军令状是「回调到 **18 度**」，
  用 `state: ["off","cool",...]` 断言时**播种值就已满足**——断言恒真，
  等于没验证（vacuous assertion）。属性形态才能真回答「跑对了吗」。
- **变量形态** `{"var": "turn_on_result.success", "op": "eq", "value": true}`（`op` 缺省 `eq`）。
"""

from __future__ import annotations

from typing import Any, Iterable, Mapping, Sequence

__all__ = ["ExpectReport", "evaluate_expects", "evaluate_graph_expects", "resolve_var"]

#: 哨兵：变量不存在（与「变量存在但值为 None」区分）
MISSING: Any = object()

#: 变量形态支持的前缀（与 `af_ir.expr` 的命名空间一致）
_VAR_PREFIXES = ("vars.", "context.", "entity.")


def resolve_var(name: str, namespace: Mapping[str, Any]) -> Any:
    """按点号路径解析变量：`turn_on_result.success` / `vars.flag`。不存在返回 `MISSING`。"""
    path = str(name or "")
    for prefix in _VAR_PREFIXES:
        if path.startswith(prefix):
            path = path[len(prefix) :]
            break
    current: Any = namespace
    for part in path.split("."):
        if isinstance(current, Mapping) and part in current:
            current = current[part]
        else:
            return MISSING
    return current


def _apply_op(op: str, actual: Any, expected: Any) -> bool | None:
    """比较运算。返回 `None` 表示「无法比较」（→ unverified，而非判失败）。"""
    if op == "eq":
        if isinstance(actual, str) or isinstance(expected, str):
            return str(actual) == str(expected)
        return actual == expected
    if op == "ne":
        inner = _apply_op("eq", actual, expected)
        return None if inner is None else (not inner)
    try:
        left, right = float(actual), float(expected)
    except (TypeError, ValueError):
        return None
    if op == "lt":
        return left < right
    if op == "lte":
        return left <= right
    if op == "gt":
        return left > right
    if op == "gte":
        return left >= right
    return None


def _state_matches(actual: Any, expected: Any) -> bool:
    """实体形态：`state` 可为字符串或候选数组（命中任一即通过）。"""
    if isinstance(expected, (list, tuple)):
        return any(str(actual) == str(x) for x in expected)
    return str(actual) == str(expected)


def _read_attribute(states: Any, entity_id: str, attribute: str) -> Any:
    """从状态源读实体属性；读不到返回 `MISSING`（→ unverified，绝不编值）。

    优先走 `snapshot([entity_id])`（与生产 `HAStateProvider` / 仿真 `FakeHA` 同一入口），
    状态源不支持快照时回退直读 `.attributes`。
    """
    snapshot_fn = getattr(states, "snapshot", None)
    if callable(snapshot_fn):
        try:
            snapshot = snapshot_fn([entity_id])
            attrs = getattr(snapshot, "attributes", None) or {}
            got = attrs.get(entity_id) or {}
            if isinstance(got, Mapping) and attribute in got:
                return got[attribute]
            return MISSING
        except Exception:  # 状态源不可达（生产网络/鉴权）→ 落回直读，最终 unverified
            pass
    direct = getattr(states, "attributes", None)
    if isinstance(direct, Mapping):
        got = direct.get(entity_id) or {}
        if isinstance(got, Mapping) and attribute in got:
            return got[attribute]
    return MISSING


class ExpectReport(dict):
    """断言报告：可直接 JSON 化的 dict（`ok` / `checked` / `passed` / `failed` / `unverified` / `items`）。"""


def evaluate_expects(
    auto: Any,
    states: Any,
    var_sources: Sequence[Mapping[str, Any]] | None = None,
) -> ExpectReport:
    """对单条自动化的 `expect` 逐条求值。

    - `states`：状态源（实现 `.get(entity_id) -> str | None`，与 `af_service.simulate` 同源）。
    - `var_sources`：若干「变量命名空间」（通常每个实例一份 `ctx.vars`）；
      变量形态断言**在任一命名空间中成立即通过**。为空表示没有活跃实例 → `unverified`。
    """
    namespaces = list(var_sources or ())
    items: list[dict[str, Any]] = []
    passed = failed = unverified = 0

    for index, raw in enumerate(auto.expects()):
        item: dict[str, Any] = {"index": index, "note": raw.get("note", "")}
        if raw.get("entity_id"):
            entity_id = str(raw["entity_id"])
            attribute = raw.get("attribute")
            if attribute:
                # ── 属性形态（v1.7.1）────────────────────────────────────
                attr = str(attribute)
                op = str(raw.get("op") or "eq")
                expected = raw.get("value")
                actual = _read_attribute(states, entity_id, attr)
                item.update(
                    {
                        "kind": "entity_attribute",
                        "target": f"{entity_id}.{attr}",
                        "op": op,
                        "expected": expected,
                        "actual": None if actual is MISSING else actual,
                    }
                )
                if actual is MISSING:
                    item.update(
                        status="unverified",
                        reason=(
                            f"实体 {entity_id} 的 {attr!r} 属性不在状态源中"
                            "（未被播种或设备未上报），本次仿真无法验证该断言"
                        ),
                    )
                    unverified += 1
                else:
                    outcome = _apply_op(op, actual, expected)
                    if outcome is True:
                        item["status"] = "pass"
                        passed += 1
                    elif outcome is None:
                        item.update(status="unverified", reason="无法比较（类型不可比）")
                        unverified += 1
                    else:
                        item["status"] = "fail"
                        item["reason"] = f"期望 {op} {expected!r}，实际 {actual!r}"
                        failed += 1
            else:
                expected = raw.get("state")
                actual = states.get(entity_id) if hasattr(states, "get") else None
                item.update(
                    {"kind": "entity", "target": entity_id, "expected": expected, "actual": actual}
                )
                if actual is None:
                    item.update(
                        status="unverified",
                        reason="实体不在状态源中（未被播种或不可达），本次仿真无法验证该断言",
                    )
                    unverified += 1
                elif _state_matches(actual, expected):
                    item["status"] = "pass"
                    passed += 1
                else:
                    item["status"] = "fail"
                    item["reason"] = f"期望 {expected!r}，实际 {actual!r}"
                    failed += 1
        else:
            var_name = str(raw.get("var", ""))
            op = str(raw.get("op") or "eq")
            expected = raw.get("value")
            values = [resolve_var(var_name, ns) for ns in namespaces]
            present = [v for v in values if v is not MISSING]
            item.update({"kind": "var", "target": var_name, "op": op, "expected": expected})

            if not namespaces:
                item.update(status="unverified", reason="没有活跃实例（仿真未产生任何实例），无法验证变量断言")
                unverified += 1
                continue
            if not present:
                item.update(status="unverified", reason=f"没有任何实例写过变量 {var_name!r}")
                unverified += 1
                continue

            outcomes = [_apply_op(op, v, expected) for v in present]
            item["actual"] = values[0] if len(values) == 1 else present
            if any(o is True for o in outcomes):
                item["status"] = "pass"
                passed += 1
            elif all(o is None for o in outcomes):
                item.update(status="unverified", reason="无法比较（类型不可比）")
                unverified += 1
            else:
                item["status"] = "fail"
                item["reason"] = f"期望 {op} {expected!r}，实际 {present!r}"
                failed += 1

        items.append(item)

    return ExpectReport(
        ok=failed == 0,
        #: `fully_verified` 比 `ok` 更严：声明过断言、且**全部验过**（无 fail 且无 unverified）。
        #: `ok=True` 只代表「没抓到反例」——把「没验到」当「验过了」就是自欺（对齐 autoflow 的 fully_verified 语义）。
        fully_verified=bool(items) and failed == 0 and unverified == 0,
        declared=len(items),
        checked=passed + failed,
        passed=passed,
        failed=failed,
        unverified=unverified,
        items=items,
    )


def evaluate_graph_expects(
    graph: Any,
    states: Any,
    var_sources: Sequence[Mapping[str, Any]] | None = None,
    unmodeled_actions: Iterable[str] | None = None,
) -> dict[str, Any]:
    """对整个 Graph 求值：按自动化聚合。**未声明 `expect` 的自动化不进入结果**。

    `unmodeled_actions`：仿真底座未建模的动作（如 vhass/FakeHA 不认识的服务）——
    有则 `fully_verified=False`（那些动作的后果根本没被验证过）。
    """
    per_auto: dict[str, Any] = {}
    total = {"declared": 0, "passed": 0, "failed": 0, "unverified": 0}
    for auto in graph:
        if not auto.expects():
            continue
        report = evaluate_expects(auto, states, var_sources)
        per_auto[auto.id] = report
        for key in total:
            total[key] += report.get(key, 0)
    unmodeled = sorted(set(unmodeled_actions or ()))
    return {
        "ok": total["failed"] == 0,
        "fully_verified": (
            total["declared"] > 0
            and total["failed"] == 0
            and total["unverified"] == 0
            and not unmodeled
        ),
        "declared": total["declared"],
        "passed": total["passed"],
        "failed": total["failed"],
        "unverified": total["unverified"],
        "unmodeled_actions": unmodeled,
        "automations": per_auto,
    }
```

# 五、§7 关键设计决策（逐条说理由）

1. **分层轴是证据强度，不是结论好坏 → `fail` 进 `verified`。** `fail` 是真跑出来的反例证据；挪去 `inferred`（没证据）或 `non_simulable`（没仿真）都会歪曲证据面。结论好坏由旧字段 `ok` / `failed` 说话，条目自带 `status="fail"`。
2. **`unverified` 全家进 `non_simulable` + `flag="yellow"`**，包括：实体不在状态源、属性读不到、类型不可比、变量没被写过、没有活跃实例。**契约外未知状态也进 non_simulable 标黄**（fail-closed，不猜语义，`reason` 注明）。
3. **`inferred` 只有两个来源**：未声明 expect 的自动化、未被断言覆盖的最终状态。被 expect 覆盖的实体不再进 `inferred`（同一条结论不双计）。
4. **覆盖集合**优先用 `graph` 的 `auto.expects()` 原始 `entity_id`（精确）；拿不到 graph 时回退 `target.rsplit(".", 1)[0]`（属性形态启发式，属性名自带点时会失准 → 见待定 (c)）。
5. **fail-open / fail-closed 边界**：
   - **fail-closed**：红线标识 `_scrub` 清洗（落盘前所有键/字符串替换 `<redacted>`）；未知断言状态 → 标黄；未建模动作 → 标黄。
   - **fail-open（只影响 inferred 填充，不影响任何判定）**：状态源探测（`items()` / `.attributes` / `.get`）异常一律降级为空；`graph` 不可迭代时视为无自动化。降级只会**少列**条目，不会伪造条目。
6. **兼容边界**：`af_expect.py` 逐字照抄（含 docstring），一行不改；`simulate` 返回 = `evaluate_graph_expects` 原字段原样展开 + 新增字段；另加 `report_version` 做版本化保险（验收 6 两条路都占）。导入做「相对 → 平铺」双兜底，因为真机包布局我读不到。`_scrub` 会重建嵌套容器（tuple 保留 tuple），内容与旧值逐字段相等，除非命中红线标识（那种情况本来就该清洗）。
7. **不做的事（刻意不做，不是遗漏）**：不改 `af_executor.py`；不实现/不透传任何脚本逃生舱；零新依赖（只用 `typing` / `re` + 同仓 `af_expect`）；`simulate` 纯函数，无副作用——题目允许「等价报告构建函数」，执行主路径明确不在本单范围，所以**没有「桩实现假装执行」的东西**，未实现的是「执行动作」本身。

## 说不清的地方（不猜，列出）

- **(a) 「三个非空分区」是否要求任意输入下都非空？** 现实现：三个分区**恒存在、恒为 list、恒可枚举**；代表性仿真（有断言 + 有未覆盖产出 + 有不可仿真项）三条都非空（`test_layer_three_sections_non_empty_and_enumerable`）。若要求任意输入（比如全绿仿真）也非空，只能造占位条目，与「不伪造、只如实降级」冲突，故未做。
- **(b) `af_service.py` / 测试文件在真机可能已存在**：本单按**追加合并**交付，不整体覆盖；若真机 `af_service.simulate` 已有别的签名（如注入执行器/回调），需按真机签名对齐，我把签名建成「报告构建点」形态（`graph, states, var_sources, unmodeled_actions, final_states`）。
- **(c) `unmodeled_actions` 的契约是纯字符串集合**（§1），所以 non_simulable 里的动作条目 `auto_id=None`，无法归属到具体自动化；要归属需扩契约到动作级来源，本单不猜。
- **(d) `fail` 条目归属 `verified`**（见决策 1）；若采信台认定 `verified` 只许放 `pass`，需把 `fail` 单列——但契约只有三分区，故按证据强度放 `verified` 并保留 `status="fail"`。
- **(e) `final_states` 的值形态**（裸状态串 vs `{state, attributes}`）两种都接（`_as_state`），透传保持原样；真机若是第三种形态需对齐。
- **(f) 报告元数据（耗时 / 时间戳 / forge 版本）未加**：不属本单目标，也不猜字段名。
