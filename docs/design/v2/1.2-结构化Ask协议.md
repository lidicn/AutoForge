def _walk(node: Mapping[str, Any]):
    """遍历表达式树，yield 所有 operand（含 var/const/fn 调用）。"""
    if not isinstance(node, Mapping):
        return
    op = node.get("op")
    if op in _LOGIC_OPS:
        for arg in node.get("args", ()):
            yield from _walk(arg)
    elif op in _CMP_OPS:
        yield from _walk_operand(node.get("left"))
        yield from _walk_operand(node.get("right"))
    elif op in _UNARY_OPS:
        yield from _walk_operand(node.get("value"))

def _walk_operand(operand: Any):
    """遍历 operand（fn 形态要深入 args，函数参数里的 var 引用不漏）。"""
    if not isinstance(operand, Mapping):
        return
    yield operand
    if "fn" in operand:
        for arg in operand.get("args", ()):
            yield from _walk_operand(arg)

            breaker_threshold if breaker_threshold is not None
            else int(os.getenv("AUTOFORGE_BREAKER_THRESHOLD", "12"))
        )
        self.breaker_window = (
            breaker_window_s if breaker_window_s is not None
            else float(os.getenv("AUTOFORGE_BREAKER_WINDOW_S", "10"))
        )
        self.breaker_cooldown = (
            breaker_cooldown_s if breaker_cooldown_s is not None
            else float(os.getenv("AUTOFORGE_BREAKER_COOLDOWN_S", "15"))
        )

        self._seen: set[tuple[str, str, str | None]] = set()
        self._seen_order: deque[tuple[str, str, str | None]] = deque()
        self._last_accepted: dict[str, float] = {}
        self._last_state: dict[str, str] = {}
        self._changes: dict[str, deque[float]] = {}
        self._open_until: dict[str, float] = {}

        self._subs: dict[str, list[Callable[[BusEvent], None]]] = {}
        self.counts: dict[str, int] = {ACCEPTED: 0, DUPLICATE: 0, THROTTLED: 0, BREAKER_BLOCKED: 0}
        #: v0.3.0：已成功发布的自定义事件（只读观测，供验证与后续服务层使用）
        self.emitted: list[BusEvent] = []

    def subscribe(self, key: str, handler: Callable[[BusEvent], None]) -> None:
        """按 `entity_id` 或事件类型（`source`）精确订阅。**不支持通配符**。"""
        self._subs.setdefault(key, []).append(handler)

    def unsubscribe_all(self) -> None:
        self._subs.clear()

    def publish(self, event: BusEvent) -> str:
        """发布事件，返回分发结果（accepted / duplicate / throttled / breaker_open）。

        ⚠️ 自定义事件（`event.event is not None`，即 `emit` 广播）是**内部信号**：
        去重键已含 `event_id`（每次都是新事件），且产生速率受外部状态变更上限约束，
        **不参与熔断/节流统计**——否则自动化联动风暴（A 开→emit→B 跟→B 状态变更→…）
        会触发熔断/节流**自伤**，在用户连按/继电器抖动时阻断合法同步。
        """
        if self._is_duplicate(event):
            self.counts[DUPLICATE] += 1
            return DUPLICATE

        now = self.clock.monotonic()

        if event.event is None:
            # 实体状态事件：去重后做熔断 + 节流（防自动化隐式循环 / 抖动）
            self._record_change(event.entity_id, now)
            if self.is_open(event.entity_id):
                self.counts[BREAKER_BLOCKED] += 1
                self._log_drop(BREAKER_BLOCKED, event)
                return BREAKER_BLOCKED
            if not self._pass_throttle(event.entity_id, now, event.state):
                self.counts[THROTTLED] += 1
                self._log_drop(THROTTLED, event)
                return THROTTLED

        self._last_accepted[event.entity_id] = now
        self._last_state[event.entity_id] = event.state
        self.counts[ACCEPTED] += 1
        self._dispatch(event)
        return ACCEPTED

    def _log_drop(self, result: str, event: "BusEvent") -> None:
        import sys

        print(
            f"[BUS-DROP] {result} {event.entity_id}={event.state!r} src={event.source}",
            file=sys.stderr,
            flush=True,
        )

    def publish_all(self, events: Iterable[BusEvent]) -> list[str]:
        return [self.publish(e) for e in events]

    def publish_custom(self, event: str, data: Mapping[str, Any] | None = None) -> str:
        """发布一条**非实体自定义事件**（v0.3.0 跨自动化事件·发布侧，IR §4.3）。

        （同一事件名在窗口内过频 → 熔断，返回 `breaker_open`）。
        """
        bus_event = BusEvent.custom(event, data)
        result = self.publish(bus_event)
        if result == ACCEPTED:
            self.emitted.append(bus_event)
        return result

        ...

    def open_entities(self) -> list[str]:

    def _is_duplicate(self, event: BusEvent) -> bool:
        key = event.dedup_key
        if key in self._seen:
            return True
        self._seen.add(key)
        self._seen_order.append(key)
        if len(self._seen_order) > self.dedup_size:

        return False

        ...

        ...

    def _dispatch(self, event: BusEvent) -> None:
        # 精确订阅：实体维度 + 事件类型维度，无通配符
        ...

    def stats(self) -> dict[str, Any]:
        ...

def replay(events: Iterable[BusEvent], bus: EventBus) -> list[str]:
    """把一批事件灌进总线（仿真/回放用）。"""

### 1.2 各 kind 的字段与应答语义（严格白名单，fail-closed）

| kind | 允许字段 | 应答 `value` 形态 | 校验 | 归一化返回 |
|---|---|---|---|---|
| `choice` | `options`（非空、元素非空、不重复） | `str` | 必须 ∈ options | 原字符串 |
| `entity` | `entity_domain`（HA domain） | `str` | 形如 `domain.object_id`；若给了 domain 必须同域 | 去空白后的 entity_id |
| `threshold` | `min_/max_/unit_` | `int\|float`（**不收数字字符串**） | 在 `[min_, max_]` 内 | `float` |
| `time_range` | `min_/max_/unit_`（分钟 0..1439） | `["HH:MM","HH:MM"]` 或 `{"start","end"}` | 两端合法、起止不得相同、须落在 `[min_,max_]` 内 | `("HH:MM","HH:MM")`（补零） |
| `text` | 无 | `str` | strip 后非空 | 原字符串 |

- **缺省即拒并告警**：`value is None` / 空串 / 空窗口 → `AskAnswerError`，运行时打 `[ASK-REJECT]` 到 stderr（沿用 `af_bus._log_drop` 的既有风格），**绝不代填默认值**。

### 1.3 控件元数据（一一对应）

```

entity      -> entity_picker  （实体选择器，携带 entity_domain）

threshold   -> slider         （滑杆，携带 min/max/unit）

```

### 1.4 收敛纪律（运行时强制）

- 一轮 = 一次"提问→作答"回合；`begin_round()` 计轮，`register_ask()` 计本轮问数。

```python
# =============================================================================
# 结构化 Ask 运行时（v2.2.0）—— 附加块
# 不动 af_runtime.py 装配主流程；本块只做「登记挂起 / 按 kind 校验应答 / 收敛纪律」，
# 恢复流程仍走既有 Executor 的挂起-恢复机制（桥接点见文件尾 TODO）。
# =============================================================================

from dataclasses import dataclass, field
from typing import Any

from .af_ir.models import (
    AskAnswer,
    AskAnswerError,
    AskNotFound,
    AskSession,
    AskSpec,
    Node,
)

MAX_ASK_ROUNDS = 3
MAX_ASKS_PER_ROUND = 2


class AskBudgetError(RuntimeError):
    """收敛纪律违约（连问 > 3 轮 / 单轮 > 2 问）。"""


@dataclass
class AskBudget:
    """收敛纪律：连问 ≤3 轮、每轮 ≤2 问，超限即抛（运行时强制，fail-closed）。"""

    max_rounds: int = MAX_ASK_ROUNDS
    max_asks_per_round: int = MAX_ASKS_PER_ROUND
    round_no: int = 0
    asked: int = 0
    total: int = 0

    def begin_round(self) -> int:
        self.round_no += 1
        if self.round_no > self.max_rounds:
            raise AskBudgetError(f"收敛纪律违约：连问超过 {self.max_rounds} 轮（第 {self.round_no} 轮）")
        self.asked = 0
        return self.round_no

    def register_ask(self) -> int:
        if self.round_no == 0:
            self.begin_round()
        self.asked += 1
        self.total += 1
        if self.asked > self.max_asks_per_round:
            raise AskBudgetError(
                f"收敛纪律违约：单轮超过 {self.max_asks_per_round} 问（第 {self.asked} 问）"
            )
        return self.asked

    def snapshot(self) -> dict[str, int]:
        return {"round": self.round_no, "asked": self.asked, "total": self.total}


@dataclass
class AskRegistry:
    """挂起 ask 索引：ask_id -> AskSession（内存实现；DB 对接留 v2.2.0 之后）。"""

    _by_id: dict[str, AskSession] = field(default_factory=dict)

    def open(self, session: AskSession) -> AskSession:
        if session.ask_id in self._by_id and self._by_id[session.ask_id].is_pending:
            raise AskAnswerError(f"ask_id 重复：{session.ask_id!r}")
        self._by_id[session.ask_id] = session
        return session

    def get(self, ask_id: str, room: str) -> AskSession:
        session = self._by_id.get(ask_id)
        if session is None or session.room != room or not session.is_pending:
            raise AskNotFound(f"无挂起 ask：ask_id={ask_id!r} room={room!r}")
        return session

    def pending(self, room: str | None = None) -> list[AskSession]:
        out = [s for s in self._by_id.values() if s.is_pending]
        if room is not None:
            out = [s for s in out if s.room == room]
        return sorted(out, key=lambda s: (s.round_no, s.ask_index, s.ask_id))

    def close(self, ask_id: str, status: str, answer: Any = None) -> AskSession:
        session = self._by_id[ask_id]
        session.status = status
        session.answer = answer
        return session


@dataclass
class AskRuntime:
    """结构化 ask 运行时：suspend / resolve / answer 三入口。"""

    budget: AskBudget = field(default_factory=AskBudget)
    registry: AskRegistry = field(default_factory=AskRegistry)

    def new_round(self) -> int:
        """上一轮已应答完毕、又要追问时调用（>3 轮抛 AskBudgetError）。"""
        return self.budget.begin_round()

    def suspend(
        self, *, ask_id: str, room: str, node: Node, prompt: str, spec: AskSpec | None = None
    ) -> AskSession:
        """登记一次挂起；收敛纪律在这里强制。spec=None 表示旧自由文本 ask。"""
        ask_no = self.budget.register_ask()
        session = AskSession(
            ask_id=ask_id,
            room=room,
            node=node,
            prompt=prompt,
            spec=spec if spec is not None else AskSpec(kind="text", prompt=prompt),
            round_no=self.budget.round_no,
            ask_index=ask_no,
            structured=spec is not None,
        )
        return self.registry.open(session)

    def resolve(self, ask_id: str | None, room: str) -> AskSession:
        """resolve_ask(ask_id, room)：返回挂起会话；房间消歧在这里。"""
        if ask_id is not None:
            return self.registry.get(ask_id, room)
        pending = self.registry.pending(room)
        if not pending:
            raise AskNotFound(f"房间 {room!r} 没有挂起的 ask")
        if len(pending) > 1:
            raise AskNotFound(
                f"房间 {room!r} 有 {len(pending)} 个挂起 ask，必须显式给 ask_id（房间消歧）"
            )
        return pending[0]

    def answer(
        self, room: str, response: str | AskAnswer, ask_id: str | None = None
    ) -> tuple[AskSession, Any]:
        """消费一次应答 -> (session, 类型化 value)。任何校验失败都 fail-closed 抛错。"""
        session = self.resolve(ask_id, room)

        if isinstance(response, AskAnswer):
            if response.kind != session.spec.kind:
                raise AskAnswerError(
                    f"应答 kind={response.kind!r} 与挂起 ask 的 kind={session.spec.kind!r} 不一致"
                )
            if session.structured:
                value = session.spec.validate_answer(response.value)
            else:
                if response.value is None or not str(response.value).strip():
                    raise AskAnswerError("应答缺省——不脑补默认值，必须落成 ask")
                value = str(response.value)
        elif isinstance(response, str):
            if session.structured and session.spec.kind != "text":
                raise AskAnswerError(
                    f"该 ask 是结构化 kind={session.spec.kind!r}，请用 AskAnswer 应答"
                    "（不接受自由文本兜底）"
                )
            value = session.spec.validate_answer(response) if session.structured else response
        else:
            raise AskAnswerError(f"应答必须是 str 或 AskAnswer，收到 {type(response).__name__}")

        self.registry.close(session.ask_id, status="answered", answer=value)
        return session, value
```

        "ask_index": session.ask_index,
        "structured": session.structured,
        "spec": session.spec.to_dict(),
        "control": ask_widget_metadata(session.spec),
    }


class AskService:
    """ask 查询服务：af_api 只做序列化，数据来自 AskRuntime.registry（鸭子类型注入）。"""

    def __init__(self, registry: Any) -> None:
        self._registry = registry

    def list_asks(self, room: str | None = None) -> list[dict[str, Any]]:
        return [ask_payload(s) for s in self._registry.pending(room)]

    def get_ask(self, ask_id: str, room: str) -> dict[str, Any]:

## 6 · §7 关键设计决策

1. **`ask` 用嵌套块**：Node 顶层 `kind` 已被节点类型占用；嵌套也便于 schema 上做按 kind 的字段白名单。
2. **`AskSpecError(ValueError)` / `AskAnswerError(ValueError)`**：满足验收 #3 的"ValueError（或 SpecError）"两种说法，调用方 `except ValueError` 都能兜住。
3. **字段严格白名单**（choice 只许 options，threshold/time_range 只许 min/max/unit，entity 只许 entity_domain，text 全禁）：错配在构造期炸，比运行期静默忽略更安全；schema 侧同步 `additionalProperties: false`。
4. **数值不隐式转换**：`"24"` 判非法，与 `expr.py` IR §14-10 红线一致（HA 状态是字符串，隐式转换是 AutoFlow v1 的血泪点）。
5. **`prompt` 允许空串**：否则旧 ask 节点（可能只有空 prompt）会在合成 `AskSpec(kind="text")` 时炸掉，破坏验收 #7。缺省拒绝只针对 **answer 的 value**。
6. **time_range 归一化为 `("HH:MM","HH:MM")`，起止不得相同**；`min_/max_` 解释为**分钟数 0..1439 的可选轴范围**，两端都必须落在其中。⚠️ 这是我做的语义决定，需求没写清——备选解释是"窗口时长上下界"或"纯展示刻度"，若真仓 IR 文档另有定义，只需改 `validate_answer` 的 time_range 分支。
7. **跨午夜窗口**：`start > end` 不在构造/校验期判定，语义沿用 `expr._fn_time_between`（end<=start 即跨午夜）；但一旦给了 `min_/max_`，两端都须在界内（有界轴上不表达跨午夜）。
8. **entity 应答形如 `domain.object_id`**，`entity_domain` 限定域；`device:` / `area:` 展开形式不在应答侧接受（那是 `Node.target_entities` 的写目标语义，不是 ask 的选中值）。
9. **fail-closed 清单**：kind 非法、字段错配、应答缺省/越界/不在 options、kind 不匹配、结构化 ask 收到自由文本、房间不匹配、收敛超限 —— 全部抛错 + `[ASK-REJECT]` stderr 告警（沿用 `af_bus._log_drop` 风格）。**fail-open 项：无**。
10. **旧自由文本 ask 是唯一放行口**：`structured=False` 时 `str` 原样透传、不做非空校验，保证 `answer(room, text)` 老调用方零行为变化。
11. **AskSession 内存态**：不碰真机下发、不做 DB；重启丢挂起 = fail-closed（重新提问），与"DB 对接留 v2.2.0"一致。
12. **收敛计数的"轮"归属**：按 `AskBudget` 实例（通常一个执行链一个）计；⚠️ 需求没定义"链"的身份（room？automation 实例 id？），默认由宿主决定每个链 new 一个 `AskRuntime`。
13. **控件 id 命名**：`select / entity_picker / timeline / slider / text_input`，与 kind 双射；前端若已有命名约定，只改 `ASK_WIDGETS` 一处。
14. **`/api/asks` 信封** `{ok, count, asks, kinds, widgets}`：`ok` 字段是否符合真仓 API 惯例未知，见待读清单。
