```
def _tokenize(line: str) -> list[str]:
    """把一行切成 token：JSON 对象/数组 / JSON 字符串 / 裸词。"""
    tokens: list[str] = []
    i, n = 0, len(line)
    while i < n:
        ch = line[i]
        if ch.isspace():
            i += 1
            continue
        if ch in "{[":
            depth = 0
            start = i
            while i < n:
                c = line[i]
                if c == '"':
                    i += 1
                    while i < n and line[i] != '"':
                        if line[i] == "\\":
                            i += 1
                        i += 1
                    i += 1
                    continue
                if c in "{[":
                    depth += 1
                elif c in "}]":
                    depth -= 1
                i += 1
                if depth == 0:
                    break
            tokens.append(line[start:i])
            continue
        if ch == '"':
            start = i
            i += 1
            while i < n and line[i] != '"':
                if line[i] == "\\":
                    i += 1
                i += 1
            i += 1
            tokens.append(line[start:i])
            continue
        start = i
        while i < n and not line[i].isspace():
            i += 1
        tokens.append(line[start:i])
    return tokens
```

```python
class StagingStore:
    """会话级 IR 暂存区，带 TTL。"""
    ...

    def __init__(self, ttl: int = 3600):
        self._items: dict[str, dict[str, Any]] = {}
        self._ttl = ttl

    def put(self, graph: Any, summary: str, resolved: dict[str, str]) -> str:
        ref = f"af:{uuid.uuid4().hex[:6]}"
        self._items[ref] = {
            "graph": graph,
            "summary": summary,
            "resolved": resolved,
            "created_at": time.time(),
        }
        return ref

    def get(self, ref: str) -> dict[str, Any]:
        self._expire()
        if ref not in self._items:
            raise DraftError("E_REF_NOT_FOUND", f"ref {ref} 不存在或已过期")

        now = time.time()
        expired = [k for k, v in self._items.items() if now - v["created_at"] > self._ttl]
        for k in expired:
            del self._items[k]
```

## 1 · 设计

### 1.1 词表的唯一真相：**不手抄，从 Schema 推导**

这是本单最关键的一个决定，先讲论证：

> `ir.schema.json` 对 `automation` 顶层和 `$defs/node` 都是 `additionalProperties: false`。
> 而 `compile_spec` 产出的字典最终必须过 `load_graph` 的 Schema 校验。
> ⇒ `_parse_node` / `_parse_option` **能写出去的 key，必然是 Schema properties 的子集**。

所以把词表定义成「Schema properties ∪ 文本面别名」，在结构上**保证不会窄于真实文法**——不可能因为词表抄漏而误拒（fail-closed 的最大风险被消掉了）。反过来，`bogus_key` 不在 Schema properties 里，必然被拒。

推导是**运行时读 `ir.schema.json` 并缓存**，而不是硬编码；`ir.schema.json` 本身一个字不改（只读）。硬编码版本只作为读不到 Schema 时的兜底，并配一条单测断言二者一致。

文本面别名目前只有一个被内联内容**证实**的：`result → result_var`（docstring 例子 `result turn_on_result`）。按「宁可宽、不可窄」策略再补一个 `confirm → requires_confirm`（多收无害：真没有就退回现状的静默忽略；真有而漏收才是回归）。


### 1.3 检查时机：**编译前、token 层、行级**

```
读行 → _tokenize → 【词表前置校验】→ 分派 automation / node / edge / option
                          ↑ 未知即抛，永远走不到 _parse_node / load_graph
```

好处：错误**带行号、带 token**，而不是被 Schema 校验器用一堆 `additionalProperties` 噪声淹没。

### 1.4 位置参数与关键字字段的消歧（不猜文法的做法）

节点行长这样：`<kind> <id> [位置参数…] [key value]…`。位置参数几个、哪些是位置参数——在 `_parse_node`（未内联）里。**我不猜**，改用**区间扫描**：

- 每个 kind 给一个位置参数个数区间 `[lo, hi]`；
- 对区间内每个切分点，检查尾部是不是 `key value` 成对、key 是标识符、key ∈ 闭集；
- **存在任一合法切分 → 放行**；
- 所有切分都撞非法 key → `E_UNKNOWN_FIELD`；
- 所有切分都结构不明（奇数个 / key 不是标识符）→ **放行**，交给 `_parse_node`。

实测推演（含验收用例）：

| 输入 | 结果 |
|---|---|
| `do d1 ha.light.turn_on {"entity_id":"x"} bogus_key 1` | 唯一可行切分撞 `bogus_key` → **拒** ✅（验收 4） |
| `do d1 ha.light.turn_on {"entity_id":"x"} result r` | 切分命中 `result`→`result_var` → **过** ✅ |
| `do d1 ha.light.turn_on result r`（params 省略） | 小切分命中 → **过** ✅ |
| `set s1 myvar 5 "x"`（尾部奇数） | 无可行切分、无违规 → **fail-open** |
| `pass p1 bogus 1` | **拒** ✅ |

### 1.5 意图侧（`draft_intent`）

意图 JSON 的真实形状**没有内联**（`draft_intent:93` / `_resolve_do:288` 都是 `...`），所以我用**递归扫描**找「动作域候选」而不是写死路径：

- 键名 `domain`（显式声明，值本身就是域）；
- 键名 `service` / `action` / `adapter` / `call` / `do` 且值是字符串（点分路径，`ha.light.turn_on → light`）；

```python
class SpecError(Exception):
    """AF-Spec 语法错误。

    封闭词表（AFSpec-forbid）要求每条错误都可机读自纠：
    ``code`` 给错误码，``fix`` 给修复提示，且 ``fix`` **恒为 dict、恒含**
    ``allowed`` / ``suggested`` 两个键，供 Agent 改完直接重投。
    """

    def __init__(
        self,
        message: str,
        *,
        code: str = "E_SPEC",
        fix: dict[str, Any] | None = None,
    ) -> None:
        self.code = code
        self.fix: dict[str, Any] = dict(fix) if fix else {}
        self.fix.setdefault("code", code)
        self.fix.setdefault("allowed", [])
        self.fix.setdefault("suggested", [])
        super().__init__(message)
```

```python
# ============ AFSpec-forbid：词表校验 ============

@lru_cache(maxsize=1)
def _schema_keys() -> tuple[frozenset[str], frozenset[str]]:
    """从 ir.schema.json 推导节点行 / 顶层行合法 key 的**权威闭集**。

    只读不写（铁律：不改 Schema）。Schema 已 `additionalProperties:false`，
    所以这个集合**结构上不会窄于真实文法**——这是「不误拒」的证明，不是赌。
    """
    try:
        from .af_ir.models import SCHEMA_PATH
        schema = json.loads(Path(SCHEMA_PATH).read_text(encoding="utf-8"))
        return (
            frozenset(schema["$defs"]["node"]["properties"]),
            frozenset(schema["properties"]),
        )
    except Exception:  # Schema 缺失/结构变化 → 退兜底，单测会抓到
        return _NODE_KEYS_FALLBACK, _TOP_KEYS_FALLBACK


def _node_field_keys() -> frozenset[str]:
    node_keys, _ = _schema_keys()
    return node_keys | frozenset(_FIELD_ALIASES)


def _top_keywords() -> frozenset[str]:
    _, top_keys = _schema_keys()
    return top_keys | {"var"}   # 文本面用 `var <name> {...}` 逐条声明 vars


def _is_struct_token(tok: str) -> bool:
    return tok[:1] in ("{", "[", '"')


def _suggest(got: str, allowed: Iterable[str]) -> list[str]:
    return difflib.get_close_matches(got, sorted(allowed), n=3, cutoff=0.5)


def _vocab_fix(
    code: str,
    got: str,
    allowed: Iterable[str],
    *,
    hint: str,
    where: dict[str, Any] | None = None,
    positional: tuple[str, ...] = (),
    extra: dict[str, Any] | None = None,
) -> dict[str, Any]:
    allowed = sorted(allowed)
    fix: dict[str, Any] = {
        "code": code,
        "got": got,
        "allowed": allowed,
        "suggested": _suggest(got, allowed),
        "positional": list(positional),
        "hint": hint,
    }
    if where is not None:
        fix["where"] = where
    if extra:
        fix.update(extra)
    return fix


def _check_line_keyword(keyword: str, lineno: int) -> None:
    """行首关键字封闭词表：不在闭集即 `E_UNKNOWN_PRIMITIVE`（编译前即拒）。"""
    if (
        keyword in _NODE_KINDS
        or keyword in _STRUCTURAL_KEYWORDS
        or keyword in _top_keywords()
    ):
        return
    raise SpecError(
        f"第 {lineno} 行：未知原语 {keyword!r}",
        code="E_UNKNOWN_PRIMITIVE",
        fix=_vocab_fix(
            "E_UNKNOWN_PRIMITIVE",
            keyword,
            _NODE_KINDS,
            hint=(
                f"行首只接受 7 种节点原语 {list(_NODE_KINDS)}、"
                f"结构性关键字 {sorted(_STRUCTURAL_KEYWORDS)}，"
                f"或顶层字段 {sorted(_top_keywords())}；"
                f"请把 {keyword!r} 替换成合法原语或删除该行"
            ),
            where={"line": lineno},
            extra={
                "allowed_top_level": sorted(_top_keywords()),
                "structural": sorted(_STRUCTURAL_KEYWORDS),
            },
        ),
    )


def _check_node_fields(kind: str, tokens: list[str], lineno: int) -> None:
    """节点行字段封闭词表校验（fail-closed on unknown key）。

    位置参数与关键字字段的分界无法从签名推出，这里用**区间扫描**消歧：
    `kind id` 之后允许 `[lo, hi]` 个位置参数，尾部必须成对 `key value` 且
    key 取自 Schema。存在任一合法切分即放行；所有切分都撞非法 key 才报
    `E_UNKNOWN_FIELD`；切分不明则放行交给 `_parse_node`（见 §7-B fail-open）。
    """
    allowed = _node_field_keys()
    lo, hi = _NODE_PAYLOAD_ARITY[kind]
    unknown: list[str] = []

    for extra in range(lo, hi + 1):
        region = tokens[2 + extra:]          # [0]=kind [1]=id，其余为 payload/fields
        if not region:
            return                            # 整行被位置参数吃光 → 合法
        if len(region) % 2:
            continue                          # 奇数个：结构不明，换切分
        keys = region[0::2]
        if any(_is_struct_token(k) or not _IDENT_RE.match(k) for k in keys):
            continue                          # key 位不是标识符：结构不明，换切分
        bad = [k for k in keys if k not in allowed]
        if not bad:
            return                            # 找到合法切分 → 放行
        unknown.append(bad[0])

    if not unknown:
        return                                # 全部切分结构不明 → fail-open

    got = unknown[0]
    raise SpecError(
        f"第 {lineno} 行：{kind} 节点出现未定义字段 {got!r}",
        code="E_UNKNOWN_FIELD",
        fix=_vocab_fix(
            "E_UNKNOWN_FIELD",
            got,
            allowed,
            positional=_NODE_POSITIONAL_KEYS.get(kind, ()),
            hint=(
                "节点行的 key 必须取自 ir.schema.json 的 $defs/node/properties"
                f"（含文本面别名 {sorted(_FIELD_ALIASES)}）；"
                f"请删除 {got!r} 或替换为合法字段"
            ),
            where={
                "line": lineno,
                "kind": kind,
                "node": tokens[1] if len(tokens) > 1 else None,
            },
        ),
    )
```

```python
def compile_spec(text: str) -> Graph:
    """把 AF-Spec 文本编译成 Graph（经 JSON IR + Schema 校验）。"""
    automations: list[dict[str, Any]] = []
    current: dict[str, Any] | None = None

    for lineno, raw_line in enumerate(text.splitlines(), start=1):
        line = raw_line.strip()
        if not line or line.startswith("#"):
            continue
        tokens = _tokenize(line)
        keyword = tokens[0]

        _check_line_keyword(keyword, lineno)          # [增] 封闭词表前置

        if keyword == "automation":
            if len(tokens) < 2:
                raise SpecError(f"第 {lineno} 行：automation 缺 id")
            # version / mode 是 Schema 必填；未显式给出时补默认，后续 option 行可覆盖。
            current = {
                "id": tokens[1],
                "ir_version": IR_VERSION,
                "version": 1,
                "mode": "single",
                "nodes": [],
                "edges": [],
            }
            automations.append(current)
            continue

        if current is None:
            raise SpecError(f"第 {lineno} 行：{keyword!r} 出现在任何 automation 之前")

        if keyword in _NODE_KINDS:
            _check_node_fields(keyword, tokens, lineno)   # [增] 字段封闭词表
            current["nodes"].append(_parse_node(keyword, tokens, lineno))
        elif keyword == "edge":
            current["edges"].append(_parse_edge(tokens, lineno))
        else:
            _parse_option(current, keyword, tokens, lineno)

    if not automations:
        raise SpecError("AF-Spec 为空：至少需要一条 automation")
    return load_graph({"automations": automations})
```

```python
#: 「动作 token」可能出现的键名（值为点分路径）
_ACTION_TOKEN_KEYS = ("service", "action", "adapter", "call", "do")

#: 这些容器内部不找域（避免把 params.action 误当域）
_NO_RECURSE_KEYS = frozenset({
    "params", "data", "payload", "body", "fields", "attributes",
    "options", "meta", "target", "entity", "entities", "expect",
})


def _domain_from_decl(token: str) -> str | None:
    """显式 `domain` 值：本身就是域；`ha.light` 归一为 `light`。"""
    parts = [p for p in token.split(".") if p]
    if parts and parts[0] in _ACTION_ADAPTERS:
        parts = parts[1:]
    return parts[0] if parts else None


def _domain_from_path(token: str) -> str | None:
    """`ha.light.turn_on` / `light.turn_on` -> `light`；单段（纯动作名）-> None。"""
    parts = [p for p in token.split(".") if p]
    if parts and parts[0] in _ACTION_ADAPTERS:
        parts = parts[1:]
    return parts[0] if len(parts) >= 2 else None


def _iter_action_tokens(node: Any, path: str = ""):
    """递归找出意图里所有「动作域候选」：(token, path, is_explicit_domain)。"""
    if isinstance(node, dict):
        for key, val in node.items():
            p = f"{path}.{key}" if path else str(key)
            if key in _NO_RECURSE_KEYS:
                continue
            if key == "domain" and isinstance(val, str) and val:
                yield val, p, True
            elif key in _ACTION_TOKEN_KEYS and isinstance(val, str) and val:
                yield val, p, False
            else:
                yield from _iter_action_tokens(val, p)
    elif isinstance(node, (list, tuple)):
        for i, val in enumerate(node):
            yield from _iter_action_tokens(val, f"{path}[{i}]")


def check_intent_domains(intent: Any) -> None:
    """意图 JSON 动作域封闭词表校验（统一「未知域即拒」，fail-closed）。

    未知域抛 `DraftError("E_UNKNOWN_DOMAIN", ...)`，`.fix` 给合法域 + 就近建议；
    判不出域的（如单段动作名 `turn_on`）不拦，交给后续 affordance 校验。
    """
    for token, path, explicit in _iter_action_tokens(intent):
        domain = _domain_from_decl(token) if explicit else _domain_from_path(token)
        if domain is None or domain in _ACTION_DOMAINS:
            continue
        allowed = sorted(_ACTION_DOMAINS)
        raise DraftError(
            "E_UNKNOWN_DOMAIN",
            f"意图 {path}：未知动作域 {domain!r}",
            {
                "code": "E_UNKNOWN_DOMAIN",
                "got": domain,
                "token": token,
                "allowed": allowed,
                "suggested": difflib.get_close_matches(domain, allowed, n=3, cutoff=0.5),
                "hint": (
                    f"动作域必须取自封闭词表；请把 {domain!r} 换成合法域"
                    f"（如 light / switch / climate）或删除该动作"
                ),
                "where": {"path": path},
            },
        )
```

## 5 · §7 关键设计决策

**A · 数据结构与索引**
- 词表用 `frozenset` + `lru_cache` 推导，O(1) 成员判定；`difflib` 只在**已经决定报错**时才跑（不在热路径）。
- 节点行校验是**行级、无状态**的，不引入任何跨行索引，因此对 `_parse_node` 零侵入。
- 意图侧用生成器递归扫，单遍 O(节点数)，不建索引——意图体小，够用。

**B · fail-open / fail-closed 边界（逐条说明）**

| 位置 | 语义 | 理由 |
|---|---|---|
| 行首未知关键字 | **fail-closed** → `E_UNKNOWN_PRIMITIVE` | 证据完整（`_NODE_KINDS` + Schema properties 均内联可见） |
| 节点行未知字段 | **fail-closed** → `E_UNKNOWN_FIELD` | 词表由 Schema 推导，**证明不会窄于真实文法** |
| 节点行「切分不明」（奇数尾部 / key 非标识符） | **fail-open** | 那是文法问题不是词表问题，交给 `_parse_node` 报更准的错 |
| **边行**（`edge … id e2 label "x"`）未知字段 | **fail-open（本单不做）** | `_parse_edge:286` 未内联，无法排除「位置 id」形态；误拒会直接打穿验收 6 |
| 意图显式 `domain` / 点分 `service`·`action` | **fail-closed** → `E_UNKNOWN_DOMAIN` | 域是确定的，闭集也在 |
| 意图单段动作名（`turn_on`） | **fail-open** | 判不出是域还是服务名，拦了就是误伤 |

**C · 与 §1 现有实现的兼容边界**
1. `SpecError("msg")` 单参调用不变（`_json`/`compile_spec` 现有调用点零改动）。
2. **行为变更**：未知关键字在「`automation` 之前」时，旧文案「出现在任何 automation 之前」→ 新 `E_UNKNOWN_PRIMITIVE`。为满足验收 3（单行输入也要报对码）必须如此。落地时 `grep -n "出现在任何 automation" tests/`，若既有断言命中则同步更新断言。
3. `DraftError` 的 `.fix` 从 `None` 归一为 dict。`grep -n "fix is None\|\.fix" tests/unit/test_af_draft.py` 同步。
4. `render_spec` / `graph_to_raw` / `af_ir/*` / `ir.schema.json` **零改动**。
5. 零新依赖：只用 `difflib` / `re` / `functools` / `pathlib` / `json`，全部标准库。
